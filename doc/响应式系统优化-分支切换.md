副作用函数中，可能会出现条件判断的情况，我们将 effect 修改成如下

```js
document.body.innerText = proxy.ok ? proxy.text : "nothing";
```

这时候 bucket 里结构就是这样的:

```text
obj
 - text
   -[effect]
 - ok
   -[effect]
```

obj.ok 默认是 true，当 proxy.ok 修改成 false ，之后再对 proxy.text 赋值时，仍然会触发此 effect，但是由于 false 时直接插值 nothing，无论对 text 怎么修改，我们都不需要触发 effect。
<br />
此时我们期望的结构应该是这样的

```text
obj
 - text
  - []
 - ok
  - [effect]
```

# 解决方案

每次 effect 执行 时，清除所有 Set（key => [effect]） 中的当前 effect，get 时再重新将 effect 添加到当前的 key => [effect] 中
比如例子中 proxy.ok = false 时，会执行 effect，此时先清除依赖，再执行 effect，effect 中读 proxy.ok，进入到 ok 的 get 拦截器，将 effect 重新插入到 ok 的 Set 中，bucket 中结构就达到了我们的预期。
<br />
添加代码：

```js
// effect func
const effect = (fn) => {
  const effectFn = () => {
    cleanup(effectFn);
    activeEffect = effectFn;
    fn();
  };
  // 给effect添加一个属性，存放收集了此effect的Set
  effectFn.deps = [];
  effectFn();
};
const cleanup = (effectFn) => {
  for (let i = 0; i < effectFn.deps.length; i++) {
    effectFn.deps[i].delete(effectFn);
  }
  // 清空
  effectFn.deps.length = 0;
};

// get拦截tarck方法修改
const track = () => {
  if (!activeEffect) return;
  let depsMap = bucket.get(target);
  if (!depsMap) {
    bucket.set(target, (depsMap = new Map()));
  }
  let deps = depsMap.get(key);
  if (!deps) {
    depsMap.set(key, (deps = new Set()));
  }
  deps.add(activeEffect);
  activeEffect.deps.push(deps);
};
```

# 完整代码

```js
import "./style.css";

const obj = {
  text: "hello",
  ok: true,
};

// 存放 target => key => effect
const bucket = new WeakMap();

let activeEffect;
const effect = (fn) => {
  const effectFn = () => {
    cleanup(effectFn);
    activeEffect = effectFn;
    fn();
  };
  effectFn.deps = [];
  effectFn();
};

const cleanup = (effectFn) => {
  for (let i = 0; i < effectFn.deps.length; i++) {
    effectFn.deps[i].delete(effectFn);
  }
  effectFn.deps.length = 0;
};

const track = (target, key) => {
  if (!activeEffect) return;
  let depsMap = bucket.get(target);
  if (!depsMap) {
    bucket.set(target, (depsMap = new Map()));
  }
  let deps = depsMap.get(key);
  if (!deps) {
    depsMap.set(key, (deps = new Set()));
  }
  deps.add(activeEffect);
  activeEffect.deps.push(deps);
};

const trigger = (target, key) => {
  const depsMap = bucket.get(target);
  if (!depsMap) return;
  const deps = depsMap.get(key);
  deps && deps.forEach((fn) => fn());
};

const proxy = new Proxy(obj, {
  get(target, key) {
    // get的时候做track，也就是把effect func添加到桶里
    track(target, key);
    return Reflect.get(target, key);
  },
  set(target, key, value) {
    Reflect.set(target, key, value);
    // set的时候做trigger,也就是执行 effect func
    trigger(target, key);
    return true;
  },
});

effect(() => {
  console.log(1);
  document.body.innerText = proxy.ok ? proxy.text : "not";
});

setTimeout(() => {
  proxy.ok = false;
  setTimeout(() => {
    proxy.text = "bye";
  }, 1000);
}, 1000);
```

# 注意事项

以上代码执行后直接死循环，因为执行顺序是

```text
- proxy.ok = false
- 进入set trigger
- deps.forEach(fn=>fn())
- effectFn 执行
- effectFn 中先 cleanup 清除了deps中此effect
- fn 执行
- 进入get track
- deps.add(activeEffect) 又将此effect添加到deps中
```

如上，形成了类似如下的代码

```js
const set = new Set([effectFn]);
set.forEach(() => {
  set.delete(effectFn);
  set.add(effectFn);
});
```

这样就会产生死循环，这个是 Set 的规范
<br />
重新建一个 Set 就能解决这个问题，代码修改如下：

```js
// trigger
const deps = depsMap.get(key);
// 添加代码
const effects = new Set(deps);
effects.forEach((fn) => fn());
// deps && deps.forEach((fn) => fn()); 删除此行
```
